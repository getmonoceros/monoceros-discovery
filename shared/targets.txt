# app it cannot read a file outside its own directory. So files that several
# skills need are kept once under shared/<plugin>/ and copied into each skill
# at the same relative path. Edit the file here, never the copy;
# scripts/sync-shared.sh copies it, and the pre-commit hook does so on commit.
#
